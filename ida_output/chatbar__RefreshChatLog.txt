void __thiscall CUIStatusBar::_RefreshChatLog(CUIStatusBar *this)
{
  int v2; // edi

  v2 = (int)CUIStatusBar::_GetFilteredChatLogCount(this) + 1 - this->m_nChatWndLineVisible;
  if ( v2 <= 2
    || this->m_pScrChatLog.p->m_nCurPos == this->m_pScrChatLog.p->m_nScrollRange - 1
    || ZAPI.timeGetTime() - this->m_dwLastScrolled > 0x1388 )
  {
    CCtrlScrollBar::SetScrollRange(this->m_pScrChatLog.p, v2);
    CCtrlScrollBar::SetCurPos(this->m_pScrChatLog.p, v2 - 1);
    CUIStatusBar::ChatLogDraw(this);
  }
  else
  {
    CCtrlScrollBar::SetScrollRange(this->m_pScrChatLog.p, v2);
    CCtrlScrollBar::SetCurPos(this->m_pScrChatLog.p, this->m_pScrChatLog.p->m_nCurPos - 1);
    CUIStatusBar::ChatLogDraw(this);
  }
}
