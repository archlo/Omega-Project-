void __thiscall CITC::Close(CITC *this)
{
  IWzGr2DLayer *m_pInterface; // eax

  CSoundMan::PlayBGM(TSingleton<CSoundMan>::ms_pInstance, nullptr, 0, 0x258u, 0x3E8u, 0);
  CStage::FadeOut(this);
  m_pInterface = this->m_pLayer.m_pInterface;
  if ( m_pInterface )
  {
    this->m_pLayer.m_pInterface = nullptr;
    m_pInterface->Release(m_pInterface);
  }
  CWnd::Destroy(&this->m_wndChar);
  CWnd::Destroy(&this->m_wndSale);
  CWnd::Destroy(&this->m_wndPurchase);
  CWnd::Destroy(&this->m_wndInventory);
  CWnd::Destroy(&this->m_wndTab);
  CWnd::Destroy(&this->m_wndSubTab);
  CWnd::Destroy(&this->m_wndList);
  CWnd::Destroy(&this->m_wndStatus);
}
