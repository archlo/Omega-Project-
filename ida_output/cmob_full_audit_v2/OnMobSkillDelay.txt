void __thiscall CMob::OnMobSkillDelay(CMob *this, CInPacket *iPacket)
{
  this->m_delaySkill.tSkillDelayTime = CInPacket::Decode4(iPacket);
  this->m_delaySkill.tSkillDelayTime += get_update_time();
  this->m_delaySkill.nSkillID = CInPacket::Decode4(iPacket);
  this->m_delaySkill.nSLV = CInPacket::Decode4(iPacket);
  this->m_delaySkill.nOption = CInPacket::Decode4(iPacket);
}
