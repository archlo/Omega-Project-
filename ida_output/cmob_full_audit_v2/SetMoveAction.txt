void __thiscall CMob::SetMoveAction(CMob *this, int nMA, int bReload)
{
  int FineMoveDirAction; // edi

  FineMoveDirAction = CMob::GetFineMoveDirAction(this, nMA);
  if ( bReload
    || FineMoveDirAction != _ZtlSecureFuse<long>(
                              this->_ZtlSecureTear_m_nMoveAction,
                              this->_ZtlSecureTear_m_nMoveAction_CS) )
  {
    this->_ZtlSecureTear_m_nMoveAction_CS = _ZtlSecureTear<long>(FineMoveDirAction, this->_ZtlSecureTear_m_nMoveAction);
    if ( _ZtlSecureFuse<long>(this->_ZtlSecureTear_m_nOneTimeAction, this->_ZtlSecureTear_m_nOneTimeAction_CS) <= -1 )
      CMob::PrepareActionLayer(this);
  }
}
