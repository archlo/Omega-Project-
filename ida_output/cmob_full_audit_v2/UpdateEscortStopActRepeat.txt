void __thiscall CMob::UpdateEscortStopActRepeat(CMob *this)
{
  int m_nEscortStopAct; // edx

  if ( this->m_tEscortStopActTime && this->m_nEscortStopAct && this->m_pTemplate->nEscortType == 1 )
  {
    if ( this->m_tEscortStopActTime >= get_update_time() )
    {
      if ( !CMob::IsOnPlayingOneTimeAction(this) )
      {
        m_nEscortStopAct = this->m_nEscortStopAct;
        if ( m_nEscortStopAct != -1 )
        {
          CMob::_ZtlSecurePut_m_nOneTimeAction(this, m_nEscortStopAct);
          CMob::PrepareActionLayer(this);
        }
      }
    }
    else
    {
      this->m_tEscortStopActTime = 0;
      this->m_nEscortStopAct = 0;
    }
  }
}
