int __thiscall CMob::CheckDamagedByMob(CMob *this, unsigned int dwTemplateID)
{
  CMobTemplate *m_pTemplate; // edx
  unsigned int *a; // eax
  unsigned int i; // ecx
  unsigned int *v6; // eax

  m_pTemplate = this->m_pTemplate;
  if ( m_pTemplate->bDamagedByMob )
  {
    a = m_pTemplate->m_aDamagedBySelectedMob.a;
    if ( !a || !*(a - 1) )
      return 1;
    for ( i = 0; ; ++i )
    {
      v6 = m_pTemplate->m_aDamagedBySelectedMob.a;
      if ( !v6 || i >= *(v6 - 1) )
        break;
      if ( v6[i] == dwTemplateID )
        return 1;
    }
  }
  return 0;
}
