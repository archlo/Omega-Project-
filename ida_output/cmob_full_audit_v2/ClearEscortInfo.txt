void __thiscall CMob::ClearEscortInfo(CMob *this)
{
  IWzVector2D *m_pInterface; // eax

  m_pInterface = this->m_pvcActive.m_pInterface;
  if ( !m_pInterface || m_pInterface == (IWzVector2D *)12 )
  {
    this->m_dwTargetMobID = 0;
  }
  else
  {
    CVecCtrlMob::ClearEscortInfo((CVecCtrlMob *)&m_pInterface[-3]);
    this->m_dwTargetMobID = 0;
  }
}
