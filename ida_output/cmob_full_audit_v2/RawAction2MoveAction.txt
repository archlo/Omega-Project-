int __thiscall CMob::RawAction2MoveAction(CMob *this, int nAction)
{
  int result; // eax

  switch ( nAction )
  {
    case 0:
      result = 1;
      break;
    case 1:
      result = 2;
      break;
    case 2:
      result = 3;
      break;
    case 3:
      result = 6;
      break;
    case 4:
      result = 8;
      break;
    case 39:
      result = 16;
      break;
    default:
      result = 4
             * (_ZtlSecureFuse<long>(
                  this->m_pTemplate->_ZtlSecureTear_nMoveAbility,
                  this->m_pTemplate->_ZtlSecureTear_nMoveAbility_CS) == 4)
             + 2;
      break;
  }
  return result;
}
