void __thiscall CMob::ShowHitEffect(CMob *this, int nDamage)
{
  int *ZtlSecureTear_m_nOneTimeAction; // edi
  int update_time; // ebx
  CMobTemplate *m_pTemplate; // eax
  int RandomHitAction; // edi
  int ActionDelay; // eax
  int m_tHitExpire; // eax
  int v9; // eax

  ZtlSecureTear_m_nOneTimeAction = this->_ZtlSecureTear_m_nOneTimeAction;
  update_time = 0;
  if ( (unsigned int)(_ZtlSecureFuse<long>(
                        this->_ZtlSecureTear_m_nOneTimeAction,
                        this->_ZtlSecureTear_m_nOneTimeAction_CS)
                    - 13) > 8
    && (unsigned int)(_ZtlSecureFuse<long>(ZtlSecureTear_m_nOneTimeAction, this->_ZtlSecureTear_m_nOneTimeAction_CS) - 22) > 0x10
    && (unsigned int)(_ZtlSecureFuse<long>(ZtlSecureTear_m_nOneTimeAction, this->_ZtlSecureTear_m_nOneTimeAction_CS) - 7) > 2 )
  {
    if ( !this->m_stat.nDoom_ || (m_pTemplate = this->m_pTemplateByDoom) == nullptr )
      m_pTemplate = this->m_pTemplate;
    if ( nDamage >= _ZtlSecureFuse<long>(
                      m_pTemplate->_ZtlSecureTear_nPushedDamage,
                      m_pTemplate->_ZtlSecureTear_nPushedDamage_CS) )
    {
      if ( !_ZtlSecureFuse<int>(this->m_pTemplate->_ZtlSecureTear_bBoss, this->m_pTemplate->_ZtlSecureTear_bBoss_CS)
        || (update_time = get_update_time(), update_time > CMob::_ZtlSecureGet_m_tLastHitExpire(this) + 10000) )
      {
        RandomHitAction = CMob::GetRandomHitAction(this);
        CMob::_ZtlSecurePut_m_nOneTimeAction(this, RandomHitAction);
        CMob::PrepareActionLayer(this);
        ActionDelay = CMob::GetActionDelay(this, RandomHitAction);
        if ( !CMob::_ZtlSecurePut_m_tHitExpire(this, update_time + ActionDelay) )
        {
          m_tHitExpire = CMob::_ZtlSecureGet_m_tHitExpire(this);
          CMob::_ZtlSecurePut_m_tHitExpire(this, m_tHitExpire + 1);
        }
        v9 = CMob::_ZtlSecureGet_m_tHitExpire(this);
        CMob::_ZtlSecurePut_m_tLastHitExpire(this, v9);
      }
    }
  }
}
