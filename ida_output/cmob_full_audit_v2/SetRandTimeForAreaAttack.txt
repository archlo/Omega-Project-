void __thiscall CMob::SetRandTimeForAreaAttack(CMob *this, int nAction)
{
  MobAttackInfo *AttackInfo; // eax
  int nType; // ecx
  unsigned int tRandDelayAttack; // ebp
  int nRange; // ebx
  ZArray<long> *p_m_aRandTimeforAreaAttack; // edi
  unsigned int v8; // esi

  if ( (unsigned int)(nAction - 13) <= 8 )
  {
    AttackInfo = CMobTemplate::GetAttackInfo(this->m_pTemplate, nAction - 13);
    nType = AttackInfo->nType;
    if ( nType == 3 || nType == 4 )
    {
      tRandDelayAttack = AttackInfo->tRandDelayAttack;
      if ( tRandDelayAttack )
      {
        nRange = AttackInfo->tzRange.nRange;
        if ( (_ZtlSecureFuse<long>(this->_ZtlSecureTear_m_nMobCtrlState, this->_ZtlSecureTear_m_nMobCtrlState_CS) == 3
           || _ZtlSecureFuse<long>(this->_ZtlSecureTear_m_nMobCtrlState, this->_ZtlSecureTear_m_nMobCtrlState_CS) == 4)
          && nRange > 0 )
        {
          p_m_aRandTimeforAreaAttack = &this->m_aRandTimeforAreaAttack;
          do
          {
            v8 = CRand32::Random(&g_rand) % tRandDelayAttack;
            --nRange;
            *ZArray<long>::InsertBefore(p_m_aRandTimeforAreaAttack, -1) = v8;
          }
          while ( nRange );
        }
      }
    }
  }
}
