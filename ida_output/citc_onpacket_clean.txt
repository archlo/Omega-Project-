void __thiscall CITC::OnPacket(CITC *this, int nType, CInPacket *iPacket)
{
  switch ( nType )
  {
    case 410:
      CITC::OnChargeParamResult((CITC *)((char *)this - 8), iPacket);
      break;
    case 411:
      CITC::OnQueryCashResult((CITC *)((char *)this - 8), iPacket);
      break;
    case 412:
      CITC::OnNormalItemResult((CITC *)((char *)this - 8), iPacket);
      break;
  }
}
